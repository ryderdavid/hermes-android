enum _ThreadState { running, waiting, done, failed }

class _Thread {
  const _Thread(this.goal, this.state, {this.tools = 0, this.age = ''});
  final String goal;
  final _ThreadState state;
  final int tools;
  final String age;
}

const _threads = [
  _Thread(
    'Audit dependency licences',
    _ThreadState.waiting,
    tools: 7,
    age: '2m',
  ),
  _Thread('Draft release notes', _ThreadState.done, tools: 12, age: '4m'),
  _Thread('Check CI matrix', _ThreadState.running, tools: 3, age: '1m'),
];

Widget _stateIcon(BuildContext context, _ThreadState state, {double size = 16}) {
  final tokens = HermesTokens.of(context);
  return switch (state) {
    _ThreadState.running => SizedBox.square(
      dimension: size,
      child: const CircularProgressIndicator(strokeWidth: 2),
    ),
    _ThreadState.waiting => Icon(
      Icons.help_outline,
      size: size + 2,
      color: tokens.warning,
    ),
    _ThreadState.done => Icon(
      Icons.check_circle_outline,
      size: size + 2,
      color: tokens.success,
    ),
    _ThreadState.failed => Icon(
      Icons.error_outline,
      size: size + 2,
      color: tokens.danger,
    ),
  };
}

String _stateLabel(_ThreadState s) => switch (s) {
  _ThreadState.running => 'Running',
  _ThreadState.waiting => 'Needs you',
  _ThreadState.done => 'Done',
  _ThreadState.failed => 'Failed',
};

/// A thread speaking to the main chat mid-run (Phase 2).

    required this.thread,
    required this.text,
    this.question = false,
  });
  final String thread;
  final String text;
  final bool question;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final scheme = theme.colorScheme;
    return Card(
      margin: const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
      color: question ? scheme.tertiaryContainer : null,
      child: Padding(
        padding: const EdgeInsets.fromLTRB(16, 12, 16, 8),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              children: [
                Icon(
                  question ? Icons.help_outline : Icons.campaign_outlined,
                  size: 18,
                  color: question
                      ? scheme.onTertiaryContainer
                      : scheme.primary,
                ),
                const SizedBox(width: 8),
                Expanded(
                  child: Text(
                    question ? '$thread asks' : '$thread · update',
                    style: theme.textTheme.labelLarge?.copyWith(
                      color: question ? scheme.onTertiaryContainer : null,
                    ),
                  ),
                ),
              ],
            ),
            const SizedBox(height: 6),
            Text(
              text,
              style: theme.textTheme.bodyMedium?.copyWith(
                color: question ? scheme.onTertiaryContainer : null,
              ),
            ),
            const SizedBox(height: 4),
            Wrap(
              spacing: 8,
              children: [
                if (question) ...[
                  FilledButton.tonal(
                    onPressed: () {},
                    child: const Text('Reply'),
                  ),
                  TextButton(
                    onPressed: () {},
                    child: const Text('Let coordinator decide'),
                  ),
                ] else
                  TextButton(
                    onPressed: () {},
                    child: const Text('Open thread'),
                  ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}

void main() {
  final screens = <String, Widget Function(BuildContext)>{
    '01_current': _current,
    '02_coordinator_chat': _coordinator,
    '03_threads_sheet': _sheet,
    '04_thread_view': _threadView,
    '05_thread_messages': _question,
    '06_thread_report': _report,
  };
  for (final variant in goldenMatrix.where((v) => _variants.contains(v.name))) {
    for (final entry in screens.entries) {
      testWidgets('mockup ${entry.key} ${variant.name}', (tester) async {
        await pumpGolden(
          tester,
          Builder(builder: entry.value),
          variant: variant,
          settle: false,
        );
        await matchGolden(
          tester,
          'mockups/coordinator/${entry.key}_${variant.name}.png',
        );
      });
    }

    testWidgets('mockup 07_coordinator_toggle ${variant.name}', (tester) async {
      await pumpGolden(tester, const _MenuHost(), variant: variant);
      await tester.tap(find.byKey(const Key('menu')));
      await tester.pumpAndSettle();
      await matchGolden(
        tester,
        'mockups/coordinator/07_coordinator_toggle_${variant.name}.png',
      );
    });

    testWidgets('mockup 08_thread_approval ${variant.name}', (tester) async {
      await pumpGolden(
        tester,
        Builder(
          builder: (context) => Stack(
            children: [
              _coordinator(context),
              Center(
                child: TextButton(
                  onPressed: () => GatewayApprovalDialog.show(
                    context,
                    request: GatewayApprovalRequest.fromEventData({
                      'description':
                          'Thread "Check CI matrix" wants to re-run the '
                          'failing Android jobs.',
                      'command': 'gh run rerun 18734 --failed',
                      'choices': ['once', 'session', 'always', 'deny'],
                    }),
                    onRespond: (_) async {},
                  ),
